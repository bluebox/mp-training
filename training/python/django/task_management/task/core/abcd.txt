from django.contrib.auth import get_user_model
from core.models import Team, TeamMember, Project, Task, TaskAssignment, TaskComment

User = get_user_model()

# 1. Create Users
admin = User.objects.create_user(username="admin1", email="admin1@example.com", password="admin123", role="admin")

lead1 = User.objects.create_user(username="lead1", email="lead1@example.com", password="lead123", role="lead")
lead2 = User.objects.create_user(username="lead2", email="lead2@example.com", password="lead123", role="lead")

member1 = User.objects.create_user(username="member1", email="member1@example.com", password="member123", role="member")
member2 = User.objects.create_user(username="member2", email="member2@example.com", password="member123", role="member")
member3 = User.objects.create_user(username="member3", email="member3@example.com", password="member123", role="member")
member4 = User.objects.create_user(username="member4", email="member4@example.com", password="member123", role="member")
member5 = User.objects.create_user(username="member5", email="member5@example.com", password="member123", role="member")

# 2. Create Teams
team_alpha = Team.objects.create(name="Alpha Team", description="Handles core platform")
team_beta = Team.objects.create(name="Beta Team", description="Handles testing and QA")

# 3. Assign Members to Teams
TeamMember.objects.create(user=lead1, team=team_alpha, role="Lead")
TeamMember.objects.create(user=member1, team=team_alpha, role="Developer")
TeamMember.objects.create(user=member2, team=team_alpha, role="Developer")
TeamMember.objects.create(user=member3, team=team_alpha, role="Tester")

TeamMember.objects.create(user=lead2, team=team_beta, role="Lead")
TeamMember.objects.create(user=member4, team=team_beta, role="Developer")
TeamMember.objects.create(user=member5, team=team_beta, role="Tester")

# 4. Create Projects
project_internal = Project.objects.create(name="Internal Portal", description="", team=team_alpha, created_by=lead1)
project_qa = Project.objects.create(name="QA Dashboard", description="", team=team_beta, created_by=lead2)

# 5. Create Tasks
task_auth = Task.objects.create(
    title="Build Auth System",
    description="",
    status="todo",
    priority="high",
    project=project_internal,
    created_by=lead1
)

task_frontend = Task.objects.create(
    title="Frontend Layout",
    description="",
    status="in_progress",
    priority="medium",
    project=project_internal,
    created_by=lead1
)

task_tests = Task.objects.create(
    title="Write Unit Tests",
    description="",
    status="todo",
    priority="medium",
    project=project_qa,
    created_by=lead2
)

task_bug_review = Task.objects.create(
    title="Bug Review",
    description="",
    status="done",
    priority="low",
    project=project_qa,
    created_by=lead2
)

# 6. Assign Users to Tasks
TaskAssignment.objects.create(task=task_auth, user=member1, role="Developer")
TaskAssignment.objects.create(task=task_frontend, user=member2, role="Developer")
TaskAssignment.objects.create(task=task_tests, user=member4, role="Tester")
TaskAssignment.objects.create(task=task_bug_review, user=member5, role="Tester")

# 7. Add Comments
TaskComment.objects.create(task=task_auth, user=member1, comment="Started working on backend auth")
